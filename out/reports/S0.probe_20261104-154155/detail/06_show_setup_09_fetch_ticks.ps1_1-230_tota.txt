   42: function Read-DotEnv([string]$Path) {

   44:   $map = @{}
   45:   foreach ($line in [IO.File]::ReadAllLines($Path, [Text.Encoding]::UTF8)) {
   46:     $t = $line.Trim()
   47:     if ($t -eq '' -or $t.StartsWith('#')) { continue }
   48:     $i = $t.IndexOf('=')
   49:     if ($i -lt 1) { continue }
   50:     $k = $t.Substring(0, $i).Trim()
   51:     $v = $t.Substring($i + 1).Trim()
   52:     if ($v.Length -ge 2 -and (($v[0] -eq '"' -and $v[-1] -eq '"') -or ($v[0] -eq "'" -and $v[-1] -eq "'"))) {
   53:       $v = $v.Substring(1, $v.Length - 2)
   54:     }
   55:     $map[$k] = $v
   56:   }
   57:   return $map
   58: }

   68: function ConvertTo-TickRow($it) {
   69:   $ts = "$($it.cntr_tm)".Trim()
   70:   if ($ts.Length -lt 14) { return $null }
   71:   $o = To-Num $it.open_pric; $h = To-Num $it.high_pric; $l = To-Num $it.low_pric
   72:   $c = To-Num $it.cur_prc;   $v = To-Num $it.trde_qty
   73:   if ($null -in @($o, $h, $l, $c, $v)) { return $null }
   74:   [pscustomobject]@{
   75:     ts = $ts.Substring(0, 14); date = $ts.Substring(0, 8); time = $ts.Substring(8, 6)
   76:     open = $o; high = $h; low = $l; close = $c; volume = $v
   77:     amount = ($o + $h + $l + $c) / 4 * $v
   78:   }
   79: }

   81: function Invoke-Kiwoom([string]$Url, [hashtable]$Headers, [hashtable]$Body) {
   82:   $bytes = [Text.Encoding]::UTF8.GetBytes(($Body | ConvertTo-Json -Compress))
   83:   for ($attempt = 1; $attempt -le 4; $attempt++) {
   84:     try {
   85:       $resp = Invoke-WebRequest -Uri $Url -Method Post -Headers $Headers -Body $bytes `
   86:               -ContentType 'application/json;charset=UTF-8' -UseBasicParsing -TimeoutSec 20
   87:       $text = $null
   88:       if ($resp.RawContentStream) {
   89:         $ms = New-Object IO.MemoryStream; $resp.RawContentStream.Position = 0; $resp.RawContentStream.CopyTo($ms)
   90:         $text = [Text.Encoding]::UTF8.GetString($ms.ToArray())
   91:       } else { $text = $resp.Content }
   92:       return [pscustomobject]@{ Headers = $resp.Headers; Text = $text; Json = ($text | ConvertFrom-Json) }
   93:     } catch {
   94:       $status = 0; $err = $_.Exception.Message
   95:       $r = $_.Exception.Response
   96:       if ($r) {
   97:         try { $status = [int]$r.StatusCode } catch {}
   98:         try { $sr = New-Object IO.StreamReader($r.GetResponseStream(), [Text.Encoding]::UTF8); $err = $sr.ReadToEnd() } catch {}
   99:       }
  100:       if ($_.ErrorDetails -and $_.ErrorDetails.Message) { $err = $_.ErrorDetails.Message }
  101:       if ($status -eq 429 -and $attempt -lt 4) { Start-Sleep -Milliseconds (1000 * $attempt); continue }
  102:       throw "HTTP $status : $err"
  103:     }
  104:   }
  105: }
