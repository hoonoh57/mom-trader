   40:   $bytes = [Text.Encoding]::UTF8.GetBytes(($Body | ConvertTo-Json -Compress))

   45:       $text = [Text.Encoding]::UTF8.GetString($ms.ToArray())

   47:     } catch {

   49:       if ($_.Exception.Response) { try { $status = [int]$_.Exception.Response.StatusCode } catch {} }

   51:       if ($status -eq 429 -and $attempt -lt 4) { Start-Sleep -Milliseconds (1000 * $attempt); continue }
   52:       throw "HTTP $status : $err"
   53:     }
   54:   }
   55: }
